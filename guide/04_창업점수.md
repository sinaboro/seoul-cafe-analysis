# 4장. 창업 추천 점수 - NumPy

## 학습 목표

- 단위가 다른 지표들을 **0~100 점**으로 바꾸는 정규화를 이해한다.
- `np.percentile`, `np.clip` 으로 극단값의 영향을 줄인다.
- 가중치로 점수를 합치고, 가중치에 따라 결론이 바뀐다는 것을 안다.

## 4-1. 왜 점수로 바꿀까

"어디에 카페를 차리면 좋을까?" 에 답하려면 여러 지표를 함께 봐야 한다. 그런데 단위가 모두 다르다.

| 지표 | 열 | 값의 범위(대략) | 좋은 방향 |
|---|---|---|---|
| 점포당 매출 | 점포당매출 | 수백만 ~ 1억 원 | 클수록 |
| 수요 | 점포당_유동인구 | 수십 ~ 수천 명 | 클수록 |
| 성장 | 매출성장률 (1년 전 같은 분기 대비) | -50 ~ +100 % | 클수록 |
| 안정 | 폐업률 | 0 ~ 10 % | **작을수록** |

이대로 더하면 숫자가 큰 매출이 모든 것을 결정한다. 그래서 각 지표를 **같은 0~100 점**으로 바꾼 뒤 더한다.

## 4-2. 정규화 - 0~100 점 만들기

```text
점수 = (값 - 최솟값) / (최댓값 - 최솟값) × 100
```

가장 작은 동이 0점, 가장 큰 동이 100점이 된다. 그런데 매출이 유난히 큰 동이 하나 있으면 나머지가 모두 0점 근처로 몰린다. 그래서 먼저 **아래 5%·위 5% 경계값으로 잘라**(clip) 둔다.

```python
low, high = np.nanpercentile(x, [5, 95])   # 아래 5%, 위 5% 지점
x = np.clip(x, low, high)                  # 그 밖의 값은 경계값으로
score = (x - low) / (high - low) * 100     # 0~100
```

폐업률처럼 **작을수록 좋은** 지표는 `100 - 점수` 로 뒤집는다.

## 4-3. 점수 코드

```python
"""
scoring.py - 카페 창업 추천 점수 계산 함수 모음

04_창업점수.py(분석)와 service/main.py(웹 서비스)가 같은 계산을 쓰도록 함수만 따로 모은 파일이다.
"""
import numpy as np
import pandas as pd

# 지표 이름 → (열 이름, 클수록 좋으면 True)
INDICATORS = {
    "매출": ("점포당매출", True),
    "수요": ("점포당_유동인구", True),
    "성장": ("매출성장률", True),
    "안정": ("폐업률", False),          # 폐업률은 낮을수록 좋다 → 점수를 뒤집는다
}
DEFAULT_WEIGHTS = {"매출": 0.35, "수요": 0.25, "성장": 0.20, "안정": 0.20}


def to_score(values, higher_is_better=True):
    """숫자 배열을 0~100 점으로 바꾼다.
    극단값 몇 개 때문에 나머지가 모두 0 근처로 몰리지 않게, 5%·95% 지점 밖은 그 값으로 자른다(clip)."""
    x = np.asarray(values, dtype=float)
    low, high = np.nanpercentile(x, [5, 95])            # 아래 5%, 위 5% 경계값
    x = np.clip(x, low, high)                           # 경계 밖은 경계값으로
    score = (x - low) / (high - low) * 100              # 최소-최대 정규화 → 0~100
    if not higher_is_better:
        score = 100 - score                             # 낮을수록 좋은 지표는 뒤집는다
    return np.nan_to_num(score, nan=50)                 # 값이 없으면 중간 점수(50)


def startup_score(df, weights):
    """지표마다 점수를 매기고 가중 평균한 '추천점수' 열을 붙여 돌려준다"""
    df = df.copy()
    total = sum(weights.values())                       # 가중치 합이 1 이 아니어도 되게 나눠 준다
    df["추천점수"] = 0.0
    for name, (col, higher) in INDICATORS.items():
        df[name + "점수"] = to_score(df[col], higher)
        df["추천점수"] += df[name + "점수"] * weights[name] / total
    return df.sort_values("추천점수", ascending=False)


def add_growth(last, q):
    """1년 전 같은 분기(예: 20262 → 20252) 매출과 비교한 성장률(%)을 붙인다"""
    latest = last["분기"].iloc[0]
    year_ago = str(int(latest[:4]) - 1) + latest[4]      # 연도에서 1을 뺀다
    prev = q[q["분기"] == year_ago][["행정동코드", "매출액"]].rename(columns={"매출액": "작년매출"})
    out = last.merge(prev, on="행정동코드", how="left")
    out["매출성장률"] = (out["매출액"] / out["작년매출"] - 1) * 100
    return out
```

"""
04_창업점수.py - NumPy 로 행정동마다 '카페 창업 추천 점수'(0~100)를 계산한다

── 전체 흐름 ──
① 최근 분기 표에 '1년 전 같은 분기 대비 매출 성장률' 을 붙인다
② 점포가 너무 적은 동(10곳 미만)은 숫자가 들쭉날쭉해서 뺀다
③ 지표 4개를 0~100 점으로 바꾼다 (극단값은 5%·95% 지점에서 자른 뒤 최소-최대 정규화)
     점포당 매출 ↑ 좋음 · 점포당 유동인구 ↑ 좋음 · 매출 성장률 ↑ 좋음 · 폐업률 ↓ 좋음(뒤집는다)
④ 가중치로 평균을 내 최종 점수를 만들고, 상위 10곳을 출력·저장한다

실행:  python analysis/04_창업점수.py
점수 계산 함수는 analysis/scoring.py 에 있고, service/main.py 도 같은 함수를 쓴다.
"""
from pathlib import Path

import matplotlib
matplotlib.use("Agg")
import matplotlib.pyplot as plt
import pandas as pd
import seaborn as sns

ROOT = Path(__file__).resolve().parent.parent
CLEAN = ROOT / "data" / "clean"
IMG = ROOT / "images"
sns.set_theme(style="whitegrid", font="Malgun Gothic")
plt.rcParams["axes.unicode_minus"] = False

from scoring import DEFAULT_WEIGHTS, add_growth, startup_score   # 계산 함수는 scoring.py 에 있다


def main():
    q = pd.read_csv(CLEAN / "커피_분기별.csv", dtype={"분기": str, "행정동코드": str})
    last = pd.read_csv(CLEAN / "커피_최근분기.csv", dtype={"분기": str, "행정동코드": str})

    df = add_growth(last, q)                            # ①
    df = df[df["점포수"] >= 10]                         # ② 점포 10곳 미만 동은 제외
    print(f"점수를 매긴 행정동: {len(df)}곳")

    ranked = startup_score(df, DEFAULT_WEIGHTS)         # ③ ④
    cols = ["자치구", "행정동", "추천점수", "매출점수", "수요점수", "성장점수", "안정점수", "점포수"]
    top = ranked[cols].head(10)
    print(top.round(1).to_string(index=False))
    ranked.to_csv(CLEAN / "창업점수.csv", index=False, encoding="utf-8-sig")

    # 상위 10곳 점수 구성(지표별 기여)을 누적 막대로 그린다
    part = top.set_index(top["자치구"] + " " + top["행정동"])
    contrib = pd.DataFrame({name: part[name + "점수"] * w for name, w in DEFAULT_WEIGHTS.items()})
    contrib.iloc[::-1].plot(kind="barh", stacked=True, figsize=(9, 5.5),
                            color=["#6F4E37", "#C08552", "#3E7C59", "#4E6E81"])
    plt.title("카페 창업 추천 점수 상위 10 행정동 (지표별 기여)")
    plt.xlabel("점수 (100점 만점)")
    plt.ylabel("")
    plt.tight_layout()
    plt.savefig(IMG / "q7_창업점수.png", dpi=130)
    print("  그래프: images/q7_창업점수.png")


if __name__ == "__main__":
    main()

## 4-4. 점수를 해석할 때 주의할 점

- **가중치가 결론을 바꾼다.** 기본값(매출 35 · 수요 25 · 성장 20 · 안정 20)은 하나의 예일 뿐이다. 6장 화면의 슬라이더로 수요를 100 으로 올리면 순위가 완전히 달라진다. 발표에서는 **왜 이 가중치를 골랐는지** 설명해야 한다.
- **점포가 적은 동은 들쭉날쭉하다.** 점포 3곳짜리 동은 한 가게의 매출이 전체를 좌우한다. 그래서 점포 10곳 미만 동(12곳)은 뺐다.
- **폐업률은 한 분기 값이다.** 여러 분기의 평균을 쓰면 더 안정적이다 (직접 해 보기 2).
- 이 점수는 **데이터로 본 후보 목록**이다. 실제 창업은 임대료, 상가 위치, 경쟁 브랜드를 함께 봐야 한다.
